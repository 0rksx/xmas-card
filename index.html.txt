<script>
  window.addEventListener('DOMContentLoaded', () => {
    setTimeout(() => {
      document.getElementById('envelopeContainer').classList.add('open');
    }, 2000);
  });

  function updateCharCount(textarea) {
    document.getElementById('currentCount').innerText = textarea.value.length;
  }

  function submitLetter() {
    const msg = document.getElementById('message').value;
    const pin = document.getElementById('pin').value;
    const gift = document.getElementById('giftOption').value;

    if (!msg) {
      alert('メッセージを入力してください。');
      return;
    }
    if (!pin || pin.length < 4) {
      alert('暗証番号4桁を入力してください。');
      return;
    }

    if (gift === 'cannele') {
      alert('Square 500円 決済ページへ移動します。');
      return;
    } else if (gift === 'set') {
      alert('Square 1,000円 決済ページへ移動します。');
      return;
    }

    const container = document.getElementById('envelopeContainer');
    container.classList.remove('open');
    container.classList.add('sealed');

    setTimeout(() => {
      document.getElementById('resultScreen').classList.add('show');
    }, 1600);
  }
</script>
